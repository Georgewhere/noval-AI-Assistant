1. 项目概述

项目名称：本地 AI 小说聊天助手

核心功能：

  • 上传小说文本到本地知识库
  • AI 基于小说内容进行对话
  • 支持角色扮演（预设+自定义）
  • 数据完全本地存储，保护隐私
  • 手机浏览器访问，随时随地使用

2. 技术架构

前端：React Native + Expo (Web)

后端：Express.js + Node.js

AI 模型：Ollama + Qwen2.5:3b

内网穿透：ngrok

部署：GitHub Pages (前端) + 本地电脑 (后端)

// 流式聊天
export const streamChat = async (
  message: string,
  systemPrompt?: string,
  model?: string,
  onChunk?: (chunk: string) => void
): Promise => {
  const serverUrl = getServerUrl();
  
  const response = await fetch(`${serverUrl}/api/chat`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ message, systemPrompt, model: model || 'qwen2.5:3b' }),
  });

  if (!response.ok) {
    throw new Error(`HTTP error! status: ${response.status}`);
  }

  const reader = response.body?.getReader();
  if (!reader) throw new Error('No reader available');

  const decoder = new TextDecoder();
  let fullContent = '';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const text = decoder.decode(value);
    const lines = text.split('\n');

    for (const line of lines) {
      if (line.startsWith('data: ')) {
        try {
          const data = JSON.parse(line.slice(6));
          if (data.content) {
            fullContent += data.content;
            onChunk?.(data.content);
          }
          if (data.done) {
            return fullContent;
          }
        } catch (e) {
          // Ignore parse errors
        }
      }
    }
  }

  return fullContent;
};

// 带知识库的聊天
export const chatWithKnowledge = async (
  message: string,
  knowledgeBase: Array<{ id: string; title: string; chunks: Array<{ content: string; keywords: string[] }> }>,
  systemPrompt?: string,
  model?: string,
  onChunk?: (chunk: string) => void
): Promise => {
  // 搜索相关段落
  const relevantChunks = searchRelevantChunks(message, knowledgeBase, 3);
  
  let contextPrompt = '';
  if (relevantChunks.length > 0) {
    contextPrompt = '\n\n以下是相关的小说内容，请参考这些内容进行回复：\n\n';
    for (const chunk of relevantChunks) {
      contextPrompt += `【${chunk.novelTitle}】\n${chunk.content}\n\n`;
    }
  }

  const fullMessage = message + contextPrompt;
  return streamChat(fullMessage, systemPrompt, model, onChunk);
};

3.3 文本处理 (textProcessor.ts)

client/services/textProcessor.ts

import { Novel, NovelChunk } from '../types';

// 中文分词（简化版）
const chineseStopWords = new Set([
  '的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一', '一个',
  '上', '也', '很', '到', '说', '要', '去', '你', '会', '着', '没有', '看', '好',
  '自己', '这', '他', '她', '它', '们', '那', '些', '什么', '怎么', '为什么',
  '吗', '呢', '吧', '啊', '呀', '哦', '嗯', '哈', '嘿', '唉', '哎',
  '但', '而', '或', '如果', '因为', '所以', '虽然', '但是', '而且', '并且',
  '可以', '可能', '应该', '必须', '需要', '能够', '已经', '正在', '将要',
]);

// 提取关键词
export const extractKeywords = (text: string, maxKeywords: number = 20): string[] => {
  // 简单的中文分词
  const words: string[] = [];
  let currentWord = '';
  
  for (let i = 0; i < text.length; i++) {
    const char = text[i];
    const code = char.charCodeAt(0);
    
    // 中文字符范围
    if (code >= 0x4e00 && code <= 0x9fff) {
      currentWord += char;
      // 每2-4个字作为一个词
      if (currentWord.length >= 2 && currentWord.length <= 4) {
        if (Math.random() > 0.5 || i === text.length - 1) {
          words.push(currentWord);
          currentWord = '';
        }
      }
    } else {
      if (currentWord.length >= 2) {
        words.push(currentWord);
      }
      currentWord = '';
    }
  }
  
  if (currentWord.length >= 2) {
    words.push(currentWord);
  }
  
  // 统计词频
  const wordCount: Record = {};
  for (const word of words) {
    if (!chineseStopWords.has(word) && word.length >= 2) {
      wordCount[word] = (wordCount[word] || 0) + 1;
    }
  }
  
  // 按词频排序，取前N个
  return Object.entries(wordCount)
    .sort((a, b) => b[1] - a[1])
    .slice(0, maxKeywords)
    .map(([word]) => word);
};

// 文本分块
export const splitIntoChunks = (text: string, chunkSize: number = 500): string[] => {
  const chunks: string[] = [];
  
  // 按段落分割
  const paragraphs = text.split(/\n\s*\n/);
  let currentChunk = '';
  
  for (const para of paragraphs) {
    const trimmed = para.trim();
    if (!trimmed) continue;
    
    if (currentChunk.length + trimmed.length > chunkSize) {
      if (currentChunk) {
        chunks.push(currentChunk.trim());
      }
      // 如果单个段落超过chunkSize，需要进一步分割
      if (trimmed.length > chunkSize) {
        const sentences = trimmed.split(/([。！？])/);
        let subChunk = '';
        for (let i = 0; i < sentences.length; i++) {
          if (subChunk.length + sentences[i].length > chunkSize) {
            if (subChunk) chunks.push(subChunk.trim());
            subChunk = sentences[i];
          } else {
            subChunk += sentences[i];
          }
        }
        if (subChunk) currentChunk = subChunk;
        else currentChunk = '';
      } else {
        currentChunk = trimmed;
      }
    } else {
      currentChunk += (currentChunk ? '\n\n' : '') + trimmed;
    }
  }
  
  if (currentChunk) {
    chunks.push(currentChunk.trim());
  }
  
  return chunks;
};

// 处理小说文本
export const processNovelText = (text: string, novelId: string, novelTitle: string): NovelChunk[] => {
  const textChunks = splitIntoChunks(text);
  
  return textChunks.map((content, index) => ({
    id: `${novelId}-chunk-${index}`,
    novelId,
    novelTitle,
    content,
    keywords: extractKeywords(content),
    index,
  }));
};

// 搜索相关段落
export const searchRelevantChunks = (
  query: string,
  novels: Novel[],
  maxResults: number = 3
): Array => {
  const queryKeywords = extractKeywords(query, 10);
  
  const results: Array = [];
  
  for (const novel of novels) {
    for (const chunk of novel.chunks) {
      let score = 0;
      
      // 计算关键词匹配分数
      for (const queryKw of queryKeywords) {
        // 完全匹配
        if (chunk.keywords.includes(queryKw)) {
          score += 2;
        }
        // 部分匹配
        for (const chunkKw of chunk.keywords) {
          if (queryKw.includes(chunkKw) || chunkKw.includes(queryKw)) {
            score += 1;
            break;
          }
        }
        // 内容匹配
        if (chunk.content.includes(queryKw)) {
          score += 1;
        }
      }
      
      if (score > 0) {
        results.push({ ...chunk, score });
      }
    }
  }
  
  // 按分数排序，取前N个
  return results
    .sort((a, b) => b.score - a.score)
    .slice(0, maxResults);
};

import * as DocumentPicker from 'expo-document-picker';
import { Screen } from '@/components/Screen';
import { FontAwesome6 } from '@expo/vector-icons';
import { getAllNovels, saveNovel, deleteNovel } from '@/services/database';
import { processNovelText } from '@/services/textProcessor';
import { Novel } from '@/types';

export default function LibraryScreen() {
  const [novels, setNovels] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isUploading, setIsUploading] = useState(false);

  const loadNovels = useCallback(async () => {
    try {
      const data = await getAllNovels();
      setNovels(data.sort((a, b) => b.createdAt - a.createdAt));
    } catch (error) {
      console.error('Failed to load novels:', error);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    loadNovels();
  }, [loadNovels]);

  const handleUpload = async () => {
    try {
      setIsUploading(true);
      
      const result = await DocumentPicker.getDocumentAsync({
        type: 'text/plain',
        copyToCacheDirectory: true,
      });
      
      if (result.canceled || !result.assets?.[0]) {
        setIsUploading(false);
        return;
      }
      
      const file = result.assets[0];
      const response = await fetch(file.uri);
      const text = await response.text();
      
      const novelId = `novel-${Date.now()}`;
      const chunks = processNovelText(text, novelId, file.name);
      
      const novel: Novel = {
        id: novelId,
        title: file.name.replace(/\.txt$/i, ''),
        content: text,
        chunks,
        createdAt: Date.now(),
      };
      
      await saveNovel(novel);
      await loadNovels();
      
      Alert.alert('成功', `已导入《${novel.title}》，共 ${chunks.length} 个段落`);
    } catch (error) {
      console.error('Upload error:', error);
      Alert.alert('错误', '导入失败，请重试');
    } finally {
      setIsUploading(false);
    }
  };

  const handleDelete = (novel: Novel) => {
    Alert.alert(
      '确认删除',
      `确定要删除《${novel.title}》吗？`,
      [
        { text: '取消', style: 'cancel' },
        {
          text: '删除',
          style: 'destructive',
          onPress: async () => {
            await deleteNovel(novel.id);
            await loadNovels();
          },
        },
      ]
    );
  };

  const renderNovel = ({ item }: { item: Novel }) => (
    
      
        
          {item.title}
          
            {item.chunks.length} 个段落
          
          
            {new Date(item.createdAt).toLocaleDateString()}
          
        
         handleDelete(item)} style={{ padding: 8 }}>
          
        
      
    
  );

  return (
    
      
        {/* 头部 */}
        
          知识库
          
            {isUploading ? (
              
            ) : (
              <>
                
                导入小说
              
            )}
          
        

        {/* 列表 */}
        {isLoading ? (
          
            
          
        ) : (
           item.id}
            ListEmptyComponent={
              
                
                
                  还没有导入任何小说
                
                
                  点击右上角"导入小说"按钮开始
                
              
            }
          />
        )}
      
    
  );
}

const express = require('express');
const cors = require('cors');
const { spawn } = require('child_process');

const app = express();
const PORT = 3001;

app.use(cors());
app.use(express.json({ limit: '10mb' }));

// 健康检查
app.get('/api/health', (req, res) => {
  res.json({ status: 'ok' });
});

// 获取模型列表
app.get('/api/models', (req, res) => {
  const { exec } = require('child_process');
  exec('ollama list', (error, stdout, stderr) => {
    if (error) {
      return res.status(500).json({ error: '无法获取模型列表' });
    }
    const models = stdout.split('\n')
      .slice(1)
      .filter(line => line.trim())
      .map(line => {
        const parts = line.split(/\s+/);
        return { name: parts[0], size: parts[1], modified: parts.slice(2).join(' ') };
      });
    res.json({ models });
  });
});

// 聊天接口
app.post('/api/chat', (req, res) => {
  const { message, systemPrompt, model } = req.body;
  
  if (!message) {
    return res.status(400).json({ error: '消息不能为空' });
  }

  const modelName = model || 'qwen2.5:3b';
  let fullPrompt = systemPrompt 
    ? `[INST] <>\n${systemPrompt}\n<>\n\n${message} [/INST]`
    : `[INST] ${message} [/INST]`;

  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');

  const ollama = spawn('ollama', ['run', modelName], { shell: true });
  ollama.stdin.write(fullPrompt + '\n');
  ollama.stdin.end();

  ollama.stdout.on('data', (data) => {
    res.write(`data: ${JSON.stringify({ content: data.toString() })}\n\n`);
  });

  ollama.on('close', () => {
    res.write(`data: ${JSON.stringify({ done: true })}\n\n`);
    res.end();
  });

  req.on('close', () => ollama.kill());
});

app.listen(PORT, () => {
  console.log('========================================');
  console.log('   AI 小说助手 服务端已启动！');
  console.log(`   http://localhost:${PORT}`);
  console.log('========================================');
});

6.2 后端部署

  1. 安装 Node.js
  2. 安装 Ollama 并下载模型：ollama pull qwen2.5:3b
  3. 安装依赖：npm install
  4. 启动服务：node src/index.js

6.3 内网穿透

  1. 注册 ngrok 账号
  2. 配置 authtoken：ngrok config add-authtoken YOUR_TOKEN
  3. 启动穿透：ngrok http 3001
  4. 复制 ngrok 提供的地址（如 https://xxx.ngrok.io）

6.4 手机使用

  1. 用手机浏览器打开前端页面
  2. 点击右上角设置
  3. 输入 ngrok 地址
